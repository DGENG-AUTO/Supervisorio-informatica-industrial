        
    def reconnection(self):
        try:
            Window.set_system_cursor('wait')
            self._modbusCliente.open() 
            Window.set_system_cursor('arrow')
            if self._modbusCliente.is_open:
                Clock.schedule_once(lambda dt: setattr(self.ids.img_con, "source", "img/con_on.png"))
                Clock.schedule_once(lambda dt: setattr(self.ids.label_con, "color", (0, 1, 0, 1)))
                Clock.schedule_once(lambda dt: setattr(self.ids.label_con, "text", "[b]RECONECTADO[/b]"))

                # S? cria a thread se ainda n?o estiver ativa
                if not self._update_thread or not self._update_thread.is_alive():
                    self._update_thread = Thread(target=self.update)
                    self._update_thread.start()
        except Exception as erro:
            print("Erro ao tentar reconectar:", erro)
            Clock.schedule_once(lambda dt: self.reconnection(), 3)
        
    def disconection(self):
        # FECHAR A CONEX?O
        self._modbusCliente.close()
        
        # Atualizar interface na thread principal
        self.ids.img_con.source = "img/con_off.png"
        self.ids.label_con.color = (1, 0, 0, 1)
        self.ids.label_con.text = "[b]DESCONECTADO[/b]"
        
        # TENTAR RECONECTAR
        Clock.schedule_once(lambda dt: self.reconnection(), 3)
        
    def start_atendimento(self,ip_digitado,porta_digitado):
        """
            M?todo que inicia a conex?o do servidor, configura o endere?o de ip
            e a porta de CLP. Al?m disso inicia um thread para leitura de dados
            e atualiza??o de imagens da interfaces gr?ficas
        """
        
        #DANDO OS VALORES PARA AS VARI?VEIS
        self._ip = ip_digitado
        self._porta = porta_digitado
        self._modbusCliente.host = self._ip
        self._modbusCliente.port = self._porta

        try:
            Window.set_system_cursor('wait')
            #CONECTANDO AO SERVIDOR
            self._modbusCliente.open() 
            Window.set_system_cursor('arrow')

            if (self._modbusCliente.is_open):
                #ATUALIZA??O DA IMAGEM DE CONECTIVIDADE
                self.ids.img_con.source = "img/con_on.png"
                self.ids.label_con.color = (0, 1, 0, 1)
                self.ids.label_con.text = "[b]CONECTADO[/b]"
                self._j_modbusPopup.dismiss()
                #IN?CIO DA PRIMEIRA THREAD
                self._update_thread = Thread(target=self.update)
                self._update_thread.start()                
            else:
                self._j_modbusPopup.set_aviso("Falha na conex?o com o servidor.") 
                    
        except Exception as erro:
            self._j_modbusPopup.set_aviso(f"ERRO: {erro.args}")  

    def update(self):
        """
            M?todo que mostrar? as rotinas de leitura de dados, atualiza??o da interface
            e inser??o de dados no banco de dados
        """
        try:
            while self._atualiza:
                try:
                    self.estado_motor = self._modbusCliente.read_coils(800, 1)
                    if self.estado_motor and self.estado_motor[0] == 1:
                        print("Motor ligado")
                        Clock.schedule_once(lambda dt: setattr(self.ids.img_con, "source", "img/con_on.png"))
                    else:
                        print("Motor desligado")
                except Exception as erro:
                    print("Erro",erro.args)
                    self.disconection()
                sleep(self._scan_time / 1000)
        except Exception as erro:
            Clock.schedule_once(lambda dt: self._j_modbusPopup.set_aviso(f"ERRO: {erro.args}"))
