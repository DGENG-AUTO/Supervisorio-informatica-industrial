if (estado == 1):
            if (self.tipo_partida_valor == 0):
                self._modbusCliente.write_single_coil(addrs, 1)
                print("Ligando motor em partida direta...")
            elif (self.tipo_partida_valor == 1):
                self._modbusCliente.write_single_coil(addrs, 1)
                print("Ligando motor em partida soft-start...")
            elif (self.tipo_partida_valor == 2):
                self._modbusCliente.write_single_coil(addrs, 1)
                print("Ligando motor em partida inversor...")
            else:
                print("Partida desconhecida!")
            self.tipo_partida_valor = None

        elif estado == 0:
            self._modbusCliente.write_single_coil(addrs, 0)
            print("Desligando motor...")
        else:
            print("Erro ao acessar motor.")
